    <div id="ui-layer">
        <div id="score-box" class="hud-box">🏆 <span id="score">0</span></div>
        <div id="lives-box" class="hud-box">❤️ <span id="lives">3</span></div>
    </div>
    
    <div id="instructions">HOLD TO SMASH!<br><span style="font-size:14px; color:#ff3366;">Avoid Black Platforms</span></div>

    <div id="game-over-screen">
        <h1>GAME OVER</h1>
        <div id="final-score">Score: 0</div>
        <div class="loading-text">Submitting to CashPro...</div>
    </div>

<script>
    let score = 0; let lives = 3; let gameActive = true;
    let isSmashing = false; let gameStarted = false;
    let isInvulnerable = false; // 🛡️ NEW: Invincibility Flag

    // --- Scene Setup ---
    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0x1a0b2e); 
    scene.fog = new THREE.Fog(0x1a0b2e, 15, 35);

    const camera = new THREE.PerspectiveCamera(65, window.innerWidth / window.innerHeight, 0.1, 100);
    camera.position.set(0, 5, 11);
    camera.lookAt(0, -1, 0);

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.shadowMap.enabled = true;
    document.body.appendChild(renderer.domElement);

    const ambientLight = new THREE.AmbientLight(0xffffff, 0.5);
    scene.add(ambientLight);
    const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
    dirLight.position.set(5, 15, 10);
    dirLight.castShadow = true;
    scene.add(dirLight);

    const pillarGeo = new THREE.CylinderGeometry(1.2, 1.2, 800, 32); 
    const pillarMat = new THREE.MeshPhongMaterial({ color: 0x222222 });
    const pillar = new THREE.Mesh(pillarGeo, pillarMat);
    pillar.castShadow = true; pillar.receiveShadow = true;
    scene.add(pillar);

    // --- Platform System ---
    let levels = [];
    let currentLevelIndex = 0;
    let levelsGenerated = 0; 
    const levelHeight = 1.8;
    const platRadius = 3.6;
    
    const safeMat = new THREE.MeshPhongMaterial({ color: 0x9c27b0, side: THREE.DoubleSide }); 
    const dangerMat = new THREE.MeshPhongMaterial({ color: 0x050505, side: THREE.DoubleSide }); 

    function createLevel(index) {
        const group = new THREE.Group();
        group.position.y = -(index * levelHeight);

        const dangerStart = Math.random() * Math.PI * 2;
        const dangerLength = Math.PI * 0.45;

        const safeGeo = new THREE.CylinderGeometry(platRadius, platRadius, 0.4, 32, 1, false, dangerStart + dangerLength, (Math.PI * 2) - dangerLength);
        const safeMesh = new THREE.Mesh(safeGeo, safeMat);
        safeMesh.name = "safe"; safeMesh.receiveShadow = true; safeMesh.castShadow = true;
        group.add(safeMesh);

        const dangerGeo = new THREE.CylinderGeometry(platRadius, platRadius, 0.4, 32, 1, false, dangerStart, dangerLength);
        const dangerMesh = new THREE.Mesh(dangerGeo, dangerMat);
        dangerMesh.name = "danger"; dangerMesh.receiveShadow = true; dangerMesh.castShadow = true;
        group.add(dangerMesh);

        group.rotation.y = index * 0.45; 
        pillar.add(group);
        levels.push(group);
    }

    function generateMoreLevels(count) {
        for (let i = 0; i < count; i++) { 
            createLevel(levelsGenerated); 
            levelsGenerated++;
        }
    }
    generateMoreLevels(100);

    const ballRadius = 0.45;
    const ballGeo = new THREE.SphereGeometry(ballRadius, 32, 32);
    const ballMat = new THREE.MeshPhongMaterial({ color: 0x00e6ff, emissive: 0x005566 });
    const ball = new THREE.Mesh(ballGeo, ballMat);
    ball.position.set(0, ballRadius + 0.2, platRadius - 0.6);
    ball.castShadow = true;
    scene.add(ball);

    // --- Physics Variables ---
    let ballVy = 0;
    const gravity = 0.015;
    const bounceForce = 0.28;
    const maxSmashSpeed = -0.34; 
    let targetPillarY = 0;

    const raycaster = new THREE.Raycaster();
    const rayDown = new THREE.Vector3(0, -1, 0);
    let particles = [];

    function shatterLevel(levelGroup) {
        levelGroup.visible = false; 
        for(let i=0; i<15; i++) {
            const pGeo = new THREE.BoxGeometry(0.3, 0.3, 0.3);
            const p = new THREE.Mesh(pGeo, safeMat);
            p.position.copy(levelGroup.position);
            p.position.y += pillar.position.y; 
            p.position.x = (Math.random() - 0.5) * 5;
            p.position.z = platRadius - 1;
            p.userData = { vx: (Math.random() - 0.5) * 0.4, vy: Math.random() * 0.4 + 0.1, vz: Math.random() * 0.4, life: 1.0 };
            scene.add(p);
            particles.push(p);
        }
    }

    // --- Controls ---
    function startSmash(e) {
        if (!gameActive) return;
        isSmashing = true;
        if (!gameStarted) { gameStarted = true; document.getElementById("instructions").style.opacity = 0; }
    }
    function stopSmash() { isSmashing = false; }
    window.addEventListener('mousedown', startSmash); window.addEventListener('mouseup', stopSmash);
    window.addEventListener('touchstart', startSmash); window.addEventListener('touchend', stopSmash);
    window.addEventListener('resize', () => {
        camera.aspect = window.innerWidth / window.innerHeight; camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
    });

    // --- Main Game Loop ---
    function animate() {
        requestAnimationFrame(animate);
        if (!gameActive) { renderer.render(scene, camera); return; }

        pillar.rotation.y += 0.012; 
        pillar.position.y += (targetPillarY - pillar.position.y) * 0.22; 

        if (levelsGenerated - currentLevelIndex < 40) generateMoreLevels(20); 

        // If invulnerable, don't allow smashing through
        if (isSmashing && !isInvulnerable) { ballVy = maxSmashSpeed; ball.scale.set(0.8, 1.2, 0.8); } 
        else { ballVy -= gravity; }
        
        ball.position.y += ballVy;

        let currentPlatWorldY = pillar.position.y - (currentLevelIndex * levelHeight);

        if (ball.position.y - ballRadius <= currentPlatWorldY + 0.2) {
            
            if (isSmashing && !isInvulnerable) {
                raycaster.set(new THREE.Vector3(ball.position.x, currentPlatWorldY + 2, ball.position.z), rayDown);
                
                let hitDanger = false;
                if (levels[currentLevelIndex]) {
                    const intersects = raycaster.intersectObjects(levels[currentLevelIndex].children);
                    if (intersects.length > 0 && intersects[0].object.name === "danger") hitDanger = true;
                }

                if (hitDanger) {
                    if (!isInvulnerable) {
                        lives--; document.getElementById("lives").innerText = lives;
                        isSmashing = false; 
                        ballVy = bounceForce * 1.3; // Extra push up
                        ball.position.y = currentPlatWorldY + ballRadius + 0.25; 
                        
                        scene.background.setHex(0x550000);
                        setTimeout(() => scene.background.setHex(0x1a0b2e), 150);
                        
                        // 🛡️ I-FRAMES MAGIC 🛡️
                        isInvulnerable = true;
                        ball.material.color.setHex(0xff3366); // Turn red to show damage
                        setTimeout(() => { 
                            isInvulnerable = false; 
                            ball.material.color.setHex(0x00e6ff); // Back to normal cyan
                        }, 500); // Safe for 0.5 seconds

                        if (lives <= 0) gameOver();
                    }
                } else {
                    shatterLevel(levels[currentLevelIndex]);
                    score += 10; document.getElementById("score").innerText = score;
                    
                    currentLevelIndex++;
                    targetPillarY = currentLevelIndex * levelHeight;
                    currentPlatWorldY = pillar.position.y - (currentLevelIndex * levelHeight);
                }
            } else {
                // Normal Bounce or bouncing while Invulnerable
                if (ballVy < 0) { 
                    ballVy = bounceForce;
                    ball.position.y = currentPlatWorldY + ballRadius + 0.2; 
                    ball.scale.set(1.4, 0.6, 1.4);
                    // Force smash to false if we bounced while invulnerable
                    if(isInvulnerable) isSmashing = false; 
                }
            }
        }

        if (!isSmashing) {
            ball.scale.x += (1 - ball.scale.x) * 0.15;
            ball.scale.y += (1 - ball.scale.y) * 0.15;
            ball.scale.z += (1 - ball.scale.z) * 0.15;
        }

        for (let i = particles.length - 1; i >= 0; i--) {
            let p = particles[i];
            p.position.x += p.userData.vx; p.position.y += p.userData.vy; p.position.z += p.userData.vz;
            p.userData.vy -= 0.015; p.userData.life -= 0.02;
            p.rotation.x += 0.2; p.rotation.y += 0.2;
            if (p.userData.life <= 0) { scene.remove(p); particles.splice(i, 1); }
        }

        renderer.render(scene, camera);
    }

    function gameOver() {
        gameActive = false;
        document.getElementById("game-over-screen").style.display = "flex";
        document.getElementById("final-score").innerText = "Score: " + score;
        setTimeout(() => { if (window.parent) window.parent.postMessage({ 'score': score, 'action': 'endTournament' }, '*'); }, 1500);
    }

    animate();
</script>
